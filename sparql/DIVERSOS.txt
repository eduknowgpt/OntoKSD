################ VERBS 

PREFIX ontoksd: <http://purl.org/ontoksd#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX bloom: <http://purl.org/edu/bloom#>

SELECT ?skill ?skillLabel ?bloomLevel ?verb
WHERE {
  ?skill a ontoksd:Skill ;
         rdfs:label ?skillLabel ;
         ontoksd:hasCognitiveLevel ?bloomLevel .

        ?verb bloom:isActionVerbFor  ?bloomLevel .
}
ORDER BY ?bloomLevel

############# COMPETENCES

PREFIX ontoksd: <http://purl.org/ontoksd#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?competence ?disposition ?knowledge ?skill 
WHERE {
  ?competence a ontoksd:Competence ;
                       rdfs:label ?competenceLabel .

  OPTIONAL {
    ?competence ontoksd:hasCompetenceElement ?disposition .
  }

  OPTIONAL {
    ?competence ontoksd:hasKnowledgeSkill ?knowledgeSkill .
    ?knowledgeSkill ontoksd:hasKnowledgeComponent ?knowledge .

    ?knowledgeSkill ontoksd:hasSkillComponent ?skill .
  }
}
