PREFIX : <http://www.semanticweb.org/edeyson/ontologies/2024/CCO#>
SELECT ?atomicCompetence ?title ?cDescrption ?knowledge ?skill 
WHERE {
    ?atomicCompetence a :AtomicCompetence .
    ?atomicCompetence :hasTitle ?cTitle .
    ?cTitle :lsString ?title .
    ?atomicCompetence :hasDescription ?cDescription .
    ?cDescription :lsString ?description .
    ?atomicCompetence :hasKnowledgeSkill ?knowledgeSkill .
    ?knowledgeSkill :hasKnowledgePart ?knowledge .
    ?knowledgeSkill :hasSkillPart ?skill .
  
}
ORDER BY ASC(?atomicCompetence)




PREFIX : <http://www.semanticweb.org/edeyson/ontologies/2024/CCO#>
SELECT ?cArea ?atomicCompetence

    ?atomicCompetence a :AtomicCompetence .
    ?atomicCompetence :hasCompetenceArea ?cArea .
    ?cArea :hasCompetenceAreaStatement ?areaStatement .
    ?areaStatement :lsString ?area.
  
}
