PREFIX ontoksd: <http://purl.org/ontoksd#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

# Inserção dos pares KnowledgeSkill (condicional)
INSERT {
  ?ks a ontoksd:KnowledgeSkill ;
      ontoksd:hasKnowledgeComponent ?k ;
      ontoksd:hasSkillComponent ?s .
}
WHERE {
  VALUES (?k ?s) {
    (ontoksd:ArchitectureAndOrganization ontoksd:creatingSkill)
    (ontoksd:DigitalDesign ontoksd:creatingSkill)
    (ontoksd:CircuitsAndElectronics ontoksd:creatingSkill)
    (ontoksd:AnalyticalAndCriticalThinking ontoksd:applyingSkill)
    (ontoksd:MathematicsAndStatistics ontoksd:applyingSkill)
    (ontoksd:ProblemSolvingAndTroubleShooting ontoksd:applyingSkill)
    (ontoksd:ResearchAndSelfStarterLearner ontoksd:applyingSkill)
  }

  FILTER NOT EXISTS {
    ?existing a ontoksd:KnowledgeSkill ;
              ontoksd:hasKnowledgeComponent ?k ;
              ontoksd:hasSkillComponent ?s .
  }

  BIND(IRI(CONCAT("http://purl.org/ontoksd#ks_", STRUUID())) AS ?ks)
}

;
