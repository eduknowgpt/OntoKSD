PREFIX ontoksd: <http://purl.org/ontoksd#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX cc2020k: <http://purl.org/cc2020/knowledge#> 
PREFIX cc2020fpk: <http://purl.org/cc2020/foundational#> 

# Inserção dos pares KnowledgeSkill (caso ainda não existam)
INSERT {
  ?ks a ontoksd:KnowledgeSkill ;
      ontoksd:hasKnowledgeComponent ?k ;
      ontoksd:hasSkillComponent ?s .
}
WHERE {
  VALUES (?k ?s) {
    (cc2020k:ComputerNetworks ontoksd:analyzingSkill)
    (cc2020k:PlatformTechnologies ontoksd:analyzingSkill)
    (cc2020fpk:AnalyticalAndCriticalThinking ontoksd:applyingSkill)
    (cc2020fpk:MathematicsAndStatistics ontoksd:applyingSkill)
    (cc2020fpk:QualityControl ontoksd:applyingSkill)
  }

  FILTER NOT EXISTS {
    ?existing a ontoksd:KnowledgeSkill ;
              ontoksd:hasKnowledgeComponent ?k ;
              ontoksd:hasSkillComponent ?s .
  }

  BIND(IRI(CONCAT("http://purl.org/ontoksd#ks_", STRUUID())) AS ?ks)
}

;

# Inserção da competência e relacionamentos
INSERT {
  ontoksd:competence_B a ontoksd:Competence ;
  	ontoksd:hasCompetenceType ontoksd:ComputingDomain ;
	ontoksd:hasStructureType ontoksd:Atomic ;
    dcterms:title "CC2020 Competence B"@en ;
    dcterms:description "Analyze and compare several networking topologies in terms of robustness, expandability, and throughput used within a cloud enterprise."@en ;
    dcterms:source "CC2020 Curriculum Guidelines"@en ;
    skos:notation "CC2020 Competence B"@en .
	
 
  # Disposições associadas
  ontoksd:competence_B ontoksd:hasCompetenceElement ontoksd:SelfDirected .
  ontoksd:competence_B ontoksd:hasCompetenceElement ontoksd:PurposeDriven .
  ontoksd:competence_B ontoksd:hasCompetenceElement ontoksd:Responsible .

  # Relacionar com os pares KnowledgeSkill correspondentes
  ?ks a ontoksd:KnowledgeSkill .
  ontoksd:competence_B ontoksd:hasKnowledgeSkill ?ks .
}
WHERE {
  ?ks a ontoksd:KnowledgeSkill ;
      ontoksd:hasKnowledgeComponent ?k ;
      ontoksd:hasSkillComponent ?s .

  VALUES (?k ?s) {
    (cc2020k:ComputerNetworks ontoksd:analyzingSkill)
    (cc2020k:PlatformTechnologies ontoksd:analyzingSkill)
    (cc2020fpk:AnalyticalAndCriticalThinking ontoksd:applyingSkill)
    (cc2020fpk:MathematicsAndStatistics ontoksd:applyingSkill)
    (cc2020fpk:QualityControl ontoksd:applyingSkill)
  }
}
